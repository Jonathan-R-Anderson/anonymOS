# IMMUTABLE_ROOTLESS Phase 0.4 -- the section-F acceptance gates.
#
# This suite deliberately does NOT require every gate to pass.  Section 0.4 asks for "a test list
# that FAILS today and must pass to claim each property", so the contract here is that the gates
# RUN and REPORT, and that the ones already earned do not silently regress.  Adding a `require`
# line for a gate is how a property gets claimed; weakening one to make the board green would
# defeat the entire point of the phase.

require  [F] === IMMUTABLE_ROOTLESS Phase 0.4 acceptance gates
require  [F] SUMMARY immutable

# All eight gates must report a verdict.  A gate that silently fails to run is worse than one that
# fails: it leaves the property unmeasured while looking fine.
require  [F] IMMUTABLE-1
require  [F] IMMUTABLE-2
require  [F] IMMUTABLE-3
require  [F] IMMUTABLE-4
require  [F] ROOTLESS-1
require  [F] ROOTLESS-2
require  [F] ROOTLESS-3
require  [F] ROOTLESS-4

# The desktop must still come up -- an acceptance probe that runs cap-table and generation
# surgery during boot is exactly the kind of change that has broken this boot before.
require  [4.1] dual-identity spawn accepted
forbid   EARLY BOOT EXCEPTION

# ── CLAIMED PROPERTIES ────────────────────────────────────────────────────────────────────────
# Section 0.4: "must pass to claim each property".  These gates are earned as of 2026-09-07, so
# they are now required and a regression fails the build rather than quietly lowering the score.
require  [F] ROOTLESS-1 no-privilege-from-uid-zero  PASS
require  [F] ROOTLESS-2 typed-admin-caps-no-god-cap  PASS
require  [F] ROOTLESS-3 unforgeable-narrowing-revocable  PASS
require  [F] ROOTLESS-4 pid1-least-privilege  PASS
require  [F] SUMMARY immutable 2/4 -> read-mostly (not immutable);  rootless 4/4 -> ROOTLESS
require  [F] IMMUTABLE-3 atomic-update-and-rollback  PASS
require  [F] IMMUTABLE-4 no-w-xor-x-pages  PASS

# The two immutable gates that do NOT hold.  Asserted as still-failing on purpose: if either
# starts passing, this suite fails and the roadmap has to be updated to claim it -- which is the
# only way a property gets claimed deliberately rather than by accident.
require  [F] IMMUTABLE-1 system-tree-read-only-and-verified  FAIL
require  [F] IMMUTABLE-2 state-split-enforced  FAIL
